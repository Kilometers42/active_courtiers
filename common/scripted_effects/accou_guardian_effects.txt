# See educate_child_interaction (on_accept)
accou_educate_child_on_accept_effect = {
	# Save scopes used by educate_child_interaction_effect
	scope:actor = { save_scope_as = guardian_liege }
	scope:recipient = { save_scope_as = ward_liege }
	scope:secondary_actor = { save_scope_as = guardian }
	scope:secondary_recipient = { save_scope_as = ward }

	if = {
		limit = { scope:ward.age >= 15 }
		custom_description_no_bullet = {
			text = ward_nearing_adulthood_tt
			object = scope:ward
		}
	}

	scope:guardian_liege = {
		send_interface_message = {
			type = event_childhood_neutral
			title = educate_child_interaction_notification
			left_icon = scope:guardian
			right_icon = scope:ward

			#accou_debug_log_effect = { MSG = accou_debug_msg_executing_educate_child_interaction_effect }
			#debug_log_scopes = yes

			educate_child_interaction_effect = yes

			show_as_tooltip = {
				if = {
					limit = { exists = scope:ward.var:ward_travelling_to_guardian }
					scope:ward = {
						custom_tooltip = ward_departs_tt
					}
				}
				else_if = {
					limit = { exists = scope:guardian.var:guardian_travelling_to_ward }
					scope:guardian = {
						custom_tooltip = guardian_departs_tt
					}
				}
				else = {
					scope:ward = {
						set_relation_guardian = scope:guardian
					}
				}
			}
		}

		accolade_ward_education_notification_effect = yes
	}
}

accou_set_alternate_education_focus_effect = {
	if = {
		limit = { has_focus = education_martial }
		if = {
			limit = { has_trait = bossy }
			set_variable = {
				name = accou_education_focus
				value = flag:education_stewardship
			}
		}
		else_if = {
			limit = { has_trait = rowdy }
			set_variable = {
				name = accou_education_focus
				value = flag:education_intrigue
			}
		}
	}
}

# Apply an arbitrary trait mapping stored in one of the ruler's focus lists.
accou_apply_custom_trait_focus_list_effect = {
	if = {
		limit = {
			NOT = { has_variable = accou_custom_trait_focus_applied }
			$RULER$ = {
				any_in_list = {
					variable = accou_education_focus_$FOCUS$_$GENDER$_traits
					scope:accou_education_child = { has_trait = prev }
				}
			}
		}
		set_focus = education_$FOCUS$
		set_variable = {
			name = accou_custom_trait_focus_applied
			value = yes
			days = 1
		}
	}
}

# Apply player-configured preferences. If several mapped traits match, focus
# order is Diplomacy, Martial, Stewardship, Intrigue, then Learning.
accou_apply_trait_education_focus_effect = {
	save_temporary_scope_as = accou_education_child
	remove_variable = accou_custom_trait_focus_applied
	if = {
		limit = { is_female = yes }
		accou_apply_custom_trait_focus_list_effect = { RULER = $RULER$ GENDER = female FOCUS = diplomacy }
		accou_apply_custom_trait_focus_list_effect = { RULER = $RULER$ GENDER = female FOCUS = martial }
		accou_apply_custom_trait_focus_list_effect = { RULER = $RULER$ GENDER = female FOCUS = stewardship }
		accou_apply_custom_trait_focus_list_effect = { RULER = $RULER$ GENDER = female FOCUS = intrigue }
		accou_apply_custom_trait_focus_list_effect = { RULER = $RULER$ GENDER = female FOCUS = learning }
	}
	else = {
		accou_apply_custom_trait_focus_list_effect = { RULER = $RULER$ GENDER = male FOCUS = diplomacy }
		accou_apply_custom_trait_focus_list_effect = { RULER = $RULER$ GENDER = male FOCUS = martial }
		accou_apply_custom_trait_focus_list_effect = { RULER = $RULER$ GENDER = male FOCUS = stewardship }
		accou_apply_custom_trait_focus_list_effect = { RULER = $RULER$ GENDER = male FOCUS = intrigue }
		accou_apply_custom_trait_focus_list_effect = { RULER = $RULER$ GENDER = male FOCUS = learning }
	}
}

# Remove the selected trait from every focus list for one gender.
accou_remove_trait_from_focus_lists_effect = {
	remove_list_variable = { name = accou_education_focus_diplomacy_$GENDER$_traits target = var:accou_education_trait_being_configured }
	remove_list_variable = { name = accou_education_focus_martial_$GENDER$_traits target = var:accou_education_trait_being_configured }
	remove_list_variable = { name = accou_education_focus_stewardship_$GENDER$_traits target = var:accou_education_trait_being_configured }
	remove_list_variable = { name = accou_education_focus_intrigue_$GENDER$_traits target = var:accou_education_trait_being_configured }
	remove_list_variable = { name = accou_education_focus_learning_$GENDER$_traits target = var:accou_education_trait_being_configured }
}

# Save a focus selected through the in-game configuration decision.
accou_set_trait_education_focus_preference_effect = {
	if = {
		limit = { OR = { var:accou_education_gender_being_configured = flag:all var:accou_education_gender_being_configured = flag:male } }
		accou_remove_trait_from_focus_lists_effect = { GENDER = male }
		add_to_variable_list = { name = accou_education_focus_$FOCUS$_male_traits target = var:accou_education_trait_being_configured }
	}
	if = {
		limit = { OR = { var:accou_education_gender_being_configured = flag:all var:accou_education_gender_being_configured = flag:female } }
		accou_remove_trait_from_focus_lists_effect = { GENDER = female }
		add_to_variable_list = { name = accou_education_focus_$FOCUS$_female_traits target = var:accou_education_trait_being_configured }
	}
	remove_variable = accou_education_trait_being_configured
	remove_variable = accou_education_gender_being_configured
	accou_apply_education_focus_preferences_to_children_effect = yes
}

# Clear the selected in-game mapping so that focus is no longer automated.
accou_clear_trait_education_focus_preference_effect = {
	if = {
		limit = { OR = { var:accou_education_gender_being_configured = flag:all var:accou_education_gender_being_configured = flag:male } }
		accou_remove_trait_from_focus_lists_effect = { GENDER = male }
	}
	if = {
		limit = { OR = { var:accou_education_gender_being_configured = flag:all var:accou_education_gender_being_configured = flag:female } }
		accou_remove_trait_from_focus_lists_effect = { GENDER = female }
	}
	remove_variable = accou_education_trait_being_configured
	remove_variable = accou_education_gender_being_configured
	accou_apply_education_focus_preferences_to_children_effect = yes
}

# Apply changed preferences immediately instead of waiting for birthdays.
accou_apply_education_focus_preferences_to_children_effect = {
	every_courtier = {
		limit = {
			is_adult = no
			age >= childhood_education_start_age
		}
		accou_apply_trait_education_focus_effect = { RULER = root }
	}
	every_warden_hostage = {
		limit = {
			is_adult = no
			age >= childhood_education_start_age
		}
		accou_apply_trait_education_focus_effect = { RULER = root }
	}
}

accou_change_education_focus_effect = {
	if = {
		limit = { has_variable = accou_education_focus }
		switch = {
			trigger = var:accou_education_focus
			flag:education_stewardship = {
				set_focus = education_stewardship
			}
			flag:education_intrigue = {
				set_focus = education_intrigue
			}
		}
	}
}
