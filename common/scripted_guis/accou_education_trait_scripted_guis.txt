accou_select_education_trait = {
	scope = character
	saved_scopes = {
		selected_trait
	}

	effect = {
		set_variable = {
			name = accou_education_trait_being_configured
			value = scope:selected_trait
		}
		var:accou_trait_search_holder ?= { remove_localized_text = accou_trait_search }
		remove_variable = accou_trait_search_holder
		remove_localized_text = accou_trait_search
		trigger_event = accou_guardian.0102
	}
}
