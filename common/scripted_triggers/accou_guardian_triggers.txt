# Is this character a valid guardian recipient
# See educate_child_interaction (is_shown, is_valid_showing_failures_only, can_be_picked)
accou_is_valid_guardian_recipient_trigger = {
	accou_is_valid_guardian_recipient_vanilla_trigger = { ACTOR = $ACTOR$ }
}

accou_is_valid_guardian_recipient_vanilla_trigger = {
	exists = $ACTOR$
	num_of_relation_ward < 2
	NOR = {
		has_character_flag = under_offer_as_guardian_flag
		exists = var:guardian_waiting_for_ward
		exists = var:character_requested_as_educator
	}
}

# Is this character a valid child recipient
# See educate_child_interaction (is_shown, is_valid_showing_failures_only, can_be_picked)
accou_is_valid_child_recipient_trigger = {
	accou_is_valid_child_recipient_vanilla_trigger = { ACTOR = $ACTOR$ }
}

accou_is_valid_child_recipient_vanilla_trigger = {
	num_of_relation_guardian = 0
	trigger_if = {
		limit = { is_hostage = yes }
		is_hostage_of = $ACTOR$
	}
	NOR = {
		has_character_flag = under_offer_as_ward_flag
		exists = var:ward_waiting_for_guardian
		exists = var:ward_travelling_to_guardian
		exists = var:hostage_travelling_to_warden
	}
}

accou_has_education_focus_trigger = {
	trigger_if = {
		limit = { has_variable = accou_education_focus }
		var:accou_education_focus = flag:$FOCUS$
	}
	trigger_else = {
		has_focus = $FOCUS$
	}
}

# Whether the selected game-rule preference for a trait changes the focus.
accou_trait_education_focus_rule_enabled_trigger = {
	OR = {
		has_game_rule = $RULE$_diplomacy
		has_game_rule = $RULE$_martial
		has_game_rule = $RULE$_stewardship
		has_game_rule = $RULE$_intrigue
		has_game_rule = $RULE$_learning
	}
}

# Used to preserve an explicit player preference when the guardian event would
# otherwise replace a martial focus for roleplaying reasons.
accou_has_trait_education_focus_preference_trigger = {
	root = { is_ai = no }
	OR = {
		has_variable = accou_custom_trait_focus_applied
		AND = {
			has_trait = intellect_good_3
			accou_trait_education_focus_rule_enabled_trigger = {
				RULE = accou_education_focus_genius
			}
		}
		AND = {
			has_trait = intellect_good_2
			accou_trait_education_focus_rule_enabled_trigger = {
				RULE = accou_education_focus_intelligent
			}
		}
		AND = {
			has_trait = intellect_good_1
			accou_trait_education_focus_rule_enabled_trigger = {
				RULE = accou_education_focus_quick
			}
		}
		AND = {
			has_trait = bossy
			accou_trait_education_focus_rule_enabled_trigger = {
				RULE = accou_education_focus_bossy
			}
		}
		AND = {
			has_trait = charming
			accou_trait_education_focus_rule_enabled_trigger = {
				RULE = accou_education_focus_charming
			}
		}
		AND = {
			has_trait = curious
			accou_trait_education_focus_rule_enabled_trigger = {
				RULE = accou_education_focus_curious
			}
		}
		AND = {
			has_trait = pensive
			accou_trait_education_focus_rule_enabled_trigger = {
				RULE = accou_education_focus_pensive
			}
		}
		AND = {
			has_trait = rowdy
			accou_trait_education_focus_rule_enabled_trigger = {
				RULE = accou_education_focus_rowdy
			}
		}
	}
}
